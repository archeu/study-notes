    import jakarta.servlet.*;          // ← 只有包名随版本变
import jakarta.servlet.http.*;

// ① 门卫：每个进楼的请求都经过（OncePerRequestFilter 保证同一请求只过一次）
@Component
@Order(1)
public class TraceFilter extends OncePerRequestFilter {
    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res,
                                    FilterChain chain) throws ServletException, IOException {
        MDC.put("traceId", UUID.randomUUID().toString());   // 放行前
        try {
            chain.doFilter(req, res);                       // 放行：后面的所有人
        } finally {
            MDC.clear();                                    // 放行后
        }
    }
}

// ② 秘书：只管进了 DispatcherServlet 的请求
@Component
public class LoginInterceptor implements HandlerInterceptor {
    @Override
    public boolean preHandle(HttpServletRequest req, HttpServletResponse res, Object handler) {
        if (req.getHeader("Authorization") != null) return true;   // 放行
        res.setStatus(401);
        return false;                                                // 拦下
    }
}

// ③ 装修清单：启动时调用一次，把秘书安排到门口
@Configuration
public class WebConfig implements WebMvcConfigurer {
    private final LoginInterceptor loginInterceptor;

    public WebConfig(LoginInterceptor loginInterceptor) {   // 注入 Bean，别 new
        this.loginInterceptor = loginInterceptor;
    }

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        registry.addInterceptor(loginInterceptor)
                .addPathPatterns("/api/**")
                .excludePathPatterns("/api/login", "/error");
    }
}

    常踩的坑

    每条都是真实项目里反复出现的。

    门卫抛的异常，@RestControllerAdvice 接不住

    门卫在公司外面。在 Filter 里自己写好错误响应。

    postHandle 改不了 @ResponseBody 的结果

    那时响应已经写出去了。改响应体、加响应头，用 ResponseBodyAdvice。

    秘书也会拦到静态资源和 /error

    Spring Boot 里它们都走 DispatcherServlet。用 excludePathPatterns 放过。

    Boot 项目里别随手加 @EnableWebMvc

    加了它，Boot 的 MVC 自动配置整体退场，静态资源映射、spring.jackson.* 设置都可能失效。实现 WebMvcConfigurer 就够了。

    new 出来的秘书，@Autowired 是 null

    addInterceptor(new X()) 里的 X 不是 Bean。把它声明成 Bean 再注入。

    @Component 门卫默认拦 /*

    要指定 URL，用 FilterRegistrationBean；要指定先后，用 @Order 或 setOrder()。

    请求体只能读一次

    门卫可以先用 HttpServletRequestWrapper 缓存下来再放行。秘书换不了 request，做不到。

    ThreadLocal 用完不清，串号

    Tomcat 线程池会复用线程。在 afterCompletion 或 finally 里 remove()。

    适用于 Servlet 栈的 Spring MVC（Spring 5–7、Spring Boot 2–4）。WebFlux 没有 Servlet，用的是 WebFilter，不在这张图里。
